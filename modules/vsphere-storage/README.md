# vsphere-storage

Manage vSphere NAS/NFS datastores and standalone VMDK virtual disks.

## Usage

```hcl
module "storage" {
  source = "../../modules/vsphere-storage"

  datacenter = "Datacenter"
  host       = "esxi-01.lab.local"

  nas_datastores = {
    nfs-backups = {
      remote_hosts = ["10.0.0.5"]
      remote_path  = "/export/backups"
    }
  }

  vmdk_disks = {
    shared-data = {
      datastore = "datastore1"
      path      = "volumes/shared-data.vmdk"
      size_gb   = 100
    }
  }
}
```
